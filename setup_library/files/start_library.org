---
- name: Start Library
  hosts: localhost
  gather_facts: no
  vars:
    target_ip: library
    podman_images:
      - hotspot.tar
      - rasbase.tar
      # - update_display.tar
      - webserver.tar
      - wiki.tar
      - music.tar
      - calibre-web.tar
    ansible_user: root
    ansible_become_user: root
    ansible_ssh_private_key_file: /root/.ssh/id_rsa
    ansible_ssh_common_args: "-o UserKnownHostsFile=/dev/null -o StrictHostKeyChecking=no"


  tasks:
    - name: start message
      shell: |
        sleep 2
        displayit library_starting.jpg
        sleep 2
      become: true

    - name: test connect
      raw: |
        whoami
        hostname
        # df -h
      register: whowhatwhere
      tags: runthis

    - name: show results
      debug:
        msg:
          - "{{ whowhatwhere }}"
      tags: runthis

    ##########################################################
    # check for usb library usb and mount it
    ##########################################################
    - name: get usb devices
      shell: |
        lsblk --json -i -o NAME,FSTYPE,LABEL
      become: true
      register: usbdev

    - name: set fact to usb
      when: 'usbdev is search("Library_USB")'
      set_fact:
        usbdisk: "{{ ([(usbdev.stdout|from_json)]|json_query('[*].blockdevices[*].children[?label==`Library_USB`].name') |flatten)[0] }}"

    - name: set fact to usb missing
      when: 'usbdev is not search("Library_USB")'
      set_fact:
        usbdisk: "Missing"

    - debug:
        msg:
        - "{{ usbdisk }}"
        - "{{ [(usbdev.stdout|from_json)]|json_query('[*].blockdevices[*].children[?label==`Library_USB`]') |flatten }}"

    - name: show missing
      when: '"Missing" in usbdisk'
      shell: |
        displayit usb_missing_verbose.jpg
      register: shellit
      become: true

    - debug: var=shellit

    - name: Wait till inserted block
      when: '"Missing" in usbdisk'
      block:
      - name: Wait till inserted
        shell: |
          lsblk -i -o NAME,FSTYPE,LABEL|grep Library_USB && echo found || echo Missing
          displayit usb_missing_verbose.jpg
          ################ format if not there
          DEVICE=`fdisk -l|grep sd|grep -v :|awk '{print $1}'`
          LIBR=`lsblk -i -o NAME,FSTYPE,LABEL|grep Library_USB|awk '{print $3}'`
          if [ "$DEVICE" != "" ]; then
            echo "Check if Library_USB was inserted by accident and don't format"
            echo $LIBR
            if [ "$LIBR" != "Library_USB" ]; then
              lsblk -o NAME,FSTYPE,LABEL,SIZE,MOUNTPOINT|grep sd
              mkfs.exfat -n Library_USB $DEVICE
              mount $DEVICE /Library
              mkdir /Library/library
              cp /root/install/setup_library/files/ReadMe.html /Library/library/ReadMe.html
              mkdir /Library/web
              mkdir /Library/wiki
              mkdir /Library/music
            fi
          fi
        register: waitshell
        ignore_errors: true
        until: '"found" in waitshell.stdout'
        delay: 5
        retries: 1000
        become: true

      - name: show starting
        when: '"found" in waitshell.stdout'
        shell: |
          displayit library_starting.jpg
        register: shellit
        become: true

    - name: mount the usb
      shell: |
        mount -t exfat /dev/{{ usbdisk }} /Library
        df -h /Library
      register: shellmount
      become: true

    - debug: var=shellmount.stdout_lines

    - name: force create incase some idiot deleted the folders or renamed
      shell: |
        mkdir /Library/library 2>/dev/null
        mkdir /Library/web 2>/dev/null
        mkdir /Library/wiki 2>/dev/null
        mkdir /Library/music 2>/dev/null
      register: shellmd
      ignore_errors: true
      become: true

    - debug: var=shellmd.stdout_lines
##############################################################
    ##############################
    # Start Containers
    ##################

    ###############################
    # hotspot container
    - name: start hotspot message
      shell: |
        sleep 2
        displayit start_01_hotspot.jpg
        sleep 2
      become: true

    - name: see if hotspot container is running
      shell: |
        podman ps |grep hotspot >/dev/null && echo Running || echo Stopped
      register: hotspot
      become: true

    - debug: var=hotspot

    - name: start hotspot container
      when: '"Stopped" in hotspot.stdout'
      shell: |
        podman run -d --net=host \
        --privileged -w /root \
        -v /dev/bus/usb:/dev/bus/usb localhost/hotspot /bin/sh -c "/usr/bin/python3 /root/hotspot.py; sleep infinity"
        sleep 5
        podman ps -a |tail -n1|awk '{print "podman logs "$1}'|sh
      register: starthotspot
      become: true

    - name: show startup of hotspot
      when: '"Stopped" in hotspot.stdout'
      debug: var=starthotspot.stdout_lines

    - name: hotspot message
      shell: |
        sleep 2
        displayit start_02_hotspot.jpg
        sleep 2
      become: true

    ###############################
    # Start Wiki if available
    - name: see if wiki is there if not dont start
      shell: |
        test -f /Library/wiki/wiki.xml && echo exist || echo absent
        ZIMS=`ls /Library/wiki/*.zim |wc -l`
        ZIMXML=`cat /Library/wiki/wiki.xml |grep -c -oP 'path="\K[^"]*'`
        if [ "$ZIMS" -gt "$ZIMXML" ]; then
          echo absent
        fi
      register: wiki_exist

    - debug: var=wiki_exist.stdout

    - name: see if zim file is there
      when: '"absent" in wiki_exist.stdout'
      block:
        - name: look for zim files
          shell: |
            ls /Library/wiki/*.zim |grep zim -c && echo "zims" || echo "nothing"
          register: zims

        - name: creating wiki xml
          shell: |
            sleep 2
            displayit start_03_create_wiki.jpg
            sleep 2
          become: true

        - name: create wiki.xml
          when: '"zim" in zims.stdout'
          shell: |
            CWIKI=`ls /Library/wiki/*.zim |sed -z 's/\n/ /g' |awk '{print "kiwix-manage /Library/wiki/wiki.xml add "$0}'`
            podman run --privileged -v /Library/wiki:/Library/wiki localhost/wiki /bin/sh -c "$CWIKI"
          register: cwiki

        - debug: var=cwiki

    - name: see if wiki is there if not dont start
      shell: |
        test -f /Library/wiki/wiki.xml && echo exist || echo absent
      register: wiki_exist

    - name: start wiki block
      when: '"absent" in wiki_exist.stdout'
      block:
      - name: wiki missing
        shell: |
          sleep 2
          displayit start_05_wiki_missing.jpg
          sleep 2
        become: true


    - name: start wiki block
      when: '"absent" not in wiki_exist.stdout'
      block:
      - name: starting wiki
        shell: |
          sleep 2
          displayit start_03_starting_wiki.jpg
          sleep 2
        become: true

      - name: see if wiki container is running
        shell: |
          podman ps |grep wiki >/dev/null && echo Running || echo Stopped
        register: wiki
        become: true

      - debug: var=wiki

      - name: start wiki container
        when: '"Stopped" in wiki.stdout'
        shell: |
          podman run -d --name wiki --ip 10.88.0.200 --privileged \
          --restart=always --replace \
          -v /Library/wiki:/wiki -p 6902:6902 \
          localhost/wiki /bin/sh -c \
          "kiwix-serve -v -p 6902 --nodatealiases --library /wiki/wiki.xml"


        register: startwiki
        become: true

      - name: show startup of wiki
        when: '"Stopped" in wiki.stdout'
        debug: var=startwiki.stdout_lines

      - name: verify wiki started
        shell: |
          curl -s localhost:6902 |grep -E 'Welcome|Wikipedia'  && echo "started"  || echo "failed"
        register: wikiverify
        ignore_errors: true
        until: '"started" in wikiverify.stdout'
        retries: 4
        delay: 30

      - name: set fact for wikiverify true
        when: '"started" in wikiverify.stdout'
        set_fact:
          wikiexist: true
      - name: set fact for wikiexist false
        when: '"failed" in wikiverify.stdout'
        set_fact:
          wikiexist: false

      - name: starting wiki
        when: wikiexist
        shell: |
          sleep 2
          displayit start_04_starting_wiki.jpg
          sleep 2
        become: true

      - name: starting wiki failed
        when: not wikiexist
        shell: |
          sleep 2
          displayit start_05_wiki_failed.jpg
          sleep 2
        become: true

    ###############################
    # Start Maps if available
    - name: see if maps is there if not dont start
      shell: |
        C=`ls /Library/maps/* |wc -l`
        if [ "$C" -gt "0" ]; then
          echo present
        fi
        if [ "$C" -eq "0" ]; then
          echo absent
        fi
      register: maps_exist

    - debug: var=maps_exist.stdout

    - name: start maps
      when:
        - '"present" in maps_exist.stdout'
      block:
        - name: start maps
          shell: |
            podman run -d --name library_maps --ip 10.88.0.212 --replace --privileged --restart unless-stopped -p 8080:8080 -v /Library/maps:/storage/maps localhost/library_maps:latest
          register: mapsstart

        - debug: var=mapsstart

        - name: wait till maps started
          shell: |
            curl -s localhost:8080|grep 'Library Maps'
          register: mapswait
          ignore_errors: true
          until: '"Maps" in mapswait.stdout'
          retries: 8
          delay: 15



    ###############################
    # Start Music if available
    - name: see if music is there if not dont start
      shell: |
        C=`ls /Library/music/* |wc -l`
        if [ "$C" -gt "0" ]; then
          echo present
        fi
        if [ "$C" -eq "0" ]; then
          echo absent
        fi
      register: music_exist

    - debug: var=music_exist.stdout

    - name: see if music database  file is there
      when: '"present" in music_exist.stdout'
      block:
        - name: look for database files
          shell: |
            ls /Library/.music_data 2>/dev/null && echo exists || echo does_not_exist
          register: musicdb

        - name: creating musicdb
          shell: |
            sleep 2
            displayit start_035_music.jpg
            sleep 2
          become: true

        - name: create music db
          when: '"does_not_exist" in musicdb.stdout'
          shell: |
            cd /
            tar xvf /root/install/setup_library/files/music_default_data.tar
            echo "extracted"
          register: musicdbinstall


        - name: start music
          when:
            - '"present" in music_exist.stdout'
            - '"exists" in musicdb.stdout or "extracted" in musicdbinstall.stdout'
          block:
            - name: start music
              shell: |
                podman volume rm --all --force
                podman run -d --restart unless-stopped --ip 10.88.0.210 --name=music --user 0 --privileged -p 9099:5082 -v /Library/music/:/music/:ro -v /Library/.music_data:/var/lms  music:latest
              register: musicstart

            - debug: var=musicstart

            - name: wait till started
              shell: |
                curl -s http://10.1.1.1:9099|grep '404'
              register: musicwait
              ignore_errors: true
              until: '"404" in musicwait.stdout'
              retries: 8
              delay: 15

            - name: music started
              shell: |
                sleep 2
                displayit start_035_music_started.jpg
                sleep 2
              become: true

    ###############################
    # Start ebook calibre server
    - name: see if ebook folder is there
      shell: |
        C=`ls /Library/calibre |wc -l`
        if [ "$C" -gt "0" ]; then
          echo present
        fi
        if [ "$C" -eq "0" ]; then
          echo absent
        fi
      register: ebook_exist

    - name: see if ebook  database  file is there if not create it
      when: '"absent" in ebook_exist.stdout'
      block:
        - name: look for database files
          shell: |
            ls /Library/calibre/books/metadata.db 2>/dev/null && echo exists || echo does_not_exist
          register: ebookdb

        - name: create ebook  db
          when: '"does_not_exist" in ebookdb.stdout'
          shell: |
            cd /
            tar xvf /root/install/setup_library/files/calibre_database.tar
            echo "extracted"
          register: ebookdbinstall

    - name: see if ebook folder is there after tar restore
      shell: |
        C=`ls /Library/calibre |wc -l`
        if [ "$C" -gt "0" ]; then
          echo present
        fi
        if [ "$C" -eq "0" ]; then
          echo absent
        fi
      register: ebook_exist

    - name: see if ebook  database  file is there if not create it
      when: '"present" in ebook_exist.stdout'
      block:


        - name: starting ebook display
          shell: |
            sleep 2
            displayit starting_ebook.jpg
            sleep 2
          become: true



        - name: start ebook
          when:
            - '"present" in ebook_exist.stdout'
          block:
            - name: check for new ebooks
              shell: |
                ls -l /Library/calibre/put_new_books_here/* |wc -l
              register: newebooks

            - name: new ebooks
              when: 'newebooks.stdout|int > 0'
              shell: |
                displayit new_ebooks.jpg
                sleep 2
              become: true

            - name: add new ebooks
              when: 'newebooks.stdout|int > 0'
              shell: |
                podman run -d --user 0 --ip 10.88.0.212 --privileged -v /Library/calibre/books:/books:rw -v /Library/calibre/put_new_books_here:/incoming:rw localhost/calibre-web /bin/bash -c "calibredb  add --with-library=/books -r --automerge=overwrite /incoming/*; rm -rvf /incoming/*"

            - name: start ebook
              shell: |
                podman run -d --user 0 --restart unless-stopped --ip 10.88.0.211 --privileged -p 8083:8083  -v /Library/calibre/books:/books:rw  localhost/calibre-web /usr/local/bin/cps
              register: ebookstart

            - debug: var=ebookstart

            - name: wait till started
              shell: |
                curl -s http://10.1.1.1:8083|grep '<title>Calibre-Web'
              register: ebookwait
              ignore_errors: true
              until: '"<title>Calibre-Web" in ebookwait.stdout'
              retries: 20
              delay: 15

            - name: ebook started
              shell: |
                sleep 2
                displayit started_ebook.jpg
                sleep 2
              become: true



    ###############################
    # Start Webserver if available
    - name: see if webserver is there if not dont start
      shell: |
        test -d /Library/library && echo exist || echo absent
      register: www_exist


    - name: start webserver block
      when: '"exist" not in www_exist.stdout'
      block:
      - name: library folder missing
        shell: |
          sleep 2
          displayit start_06_missing.jpg
          sleep 2
        become: true

    - name: start www block
      when: '"exist" in www_exist.stdout'
      block:
      - name: starting webserver
        shell: |
          sleep 2
          displayit start_06_starting_webserver.jpg
          sleep 2
        become: true

      - name: see if www container is running
        shell: |
          podman ps |grep webserver >/dev/null && echo Running || echo Stopped
        register: www
        become: true

      - debug: var=www

      - name: start www container
        when: '"Stopped" in www.stdout'
        shell: |
          podman run -d -v /Library:/library \
          -p 80:80 \
          --ip 10.88.0.201  --privileged  \
          localhost/webserver /bin/sh -c '/usr/bin/python3 /root/library.py;sleep infinity'
        register: startwww
        become: true

      - name: show startup of www
        when: '"Stopped" in www.stdout'
        debug: var=startwww.stdout_lines
    #################################################
      - name: verify www started just index
        shell: |
          curl -s localhost|wc -l
        register: wwwindex
        ignore_errors: true
        become: true
        until: 'wwwindex.stdout|int > 10'
        delay: 5
        retries: 1000
        # become: true

      - name: library building files
        shell: |
          sleep 2
          displayit start_06_building.jpg
          sleep 2
        become: true

    #################################################


      - name: verify www started
        shell: |
          curl -s localhost|grep Shutdown -c
        register: wwwverify
        ignore_errors: true
        until: '"1" in wwwverify.stdout'
        delay: 5
        retries: 1000
        become: true

      - name: library started
        shell: |
          sleep 2
          displayit start_07_starting_webserver.jpg
          sleep 2
        become: true

      - name: final message
        shell: |
          sleep 2
          displayit library_main.jpg
          sleep 2
        become: true
